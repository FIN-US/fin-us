# DART Disclosure MCP Integration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add official OpenDART disclosure signals through a new `mcp-dart`, wire them into backend periodic monitoring, and expose the same data to NAT agents.

**Architecture:** `mcp-dart` is the only DART data boundary and uses official OpenDART APIs only. Backend treats DART disclosures as another `SignalSource`, reusing Redis hash/lock/cooldown behavior. NAT registers `finus_disclosure_signal` using the existing vendored MCP helper pattern.

**Tech Stack:** Node.js MCP SDK, OpenDART HTTP APIs, FastAPI, APScheduler, Redis scheduler state, Python NAT function registry, pytest.

```js
import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";
import AdmZip from "adm-zip";
import { XMLParser } from "fast-xml-parser";
import { Server } from "@modelcontextprotocol/sdk/server/index.js";
import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio.js";
import {
  CallToolRequestSchema,
  ListToolsRequestSchema,
} from "@modelcontextprotocol/sdk/types.js";

console.log = console.error;

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);
const rootEnvPath = path.join(__dirname, "..", ".env");
const DATA_DIR = path.join(__dirname, "data");
const CORP_CACHE_PATH = path.join(DATA_DIR, "corp-codes.json");
const CORP_CACHE_TTL_MS = 24 * 60 * 60 * 1000;
const REQUEST_TIMEOUT_MS = 8000;
const DART_BASE_URL = "https://opendart.fss.or.kr/api";
const SHAREHOLDING_DETAIL_TYPES = ["D001", "D002", "D005"];

function loadRootEnv() {
  if (!fs.existsSync(rootEnvPath)) return;
  const envText = fs.readFileSync(rootEnvPath, "utf8");
  for (const line of envText.split(/\r?\n/)) {
    const trimmed = line.trim();
    if (!trimmed || trimmed.startsWith("#")) continue;
    const eqIndex = trimmed.indexOf("=");
    if (eqIndex === -1) continue;
    const key = trimmed.slice(0, eqIndex).trim();
    let value = trimmed.slice(eqIndex + 1).trim();
    if (
      (value.startsWith('"') && value.endsWith('"')) ||
      (value.startsWith("'") && value.endsWith("'"))
    ) {
      value = value.slice(1, -1);
    }
    if (!process.env[key]) process.env[key] = value;
  }
}

loadRootEnv();

const DART_API_KEY = process.env.DART_API_KEY;

function isMissingCredential(value) {
  return !value || value.startsWith("your_") || value.includes("_here");
}

function requireDartCredential() {
  if (isMissingCredential(DART_API_KEY)) {
    throw new Error("DART_API_KEY가 설정되지 않았습니다. 루트 .env 파일을 확인하세요.");
  }
}

function normalizeStockInput(value) {
  return String(value ?? "").trim();
}

async function fetchWithTimeout(url, { responseType = "json" } = {}) {
  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), REQUEST_TIMEOUT_MS);
  try {
    const response = await fetch(url, { signal: controller.signal });
    if (!response.ok) {
      throw new Error(`OpenDART API 오류(${response.status}): ${response.statusText}`);
    }
    if (responseType === "arrayBuffer") return Buffer.from(await response.arrayBuffer());
    const payload = await response.json();
    if (payload?.status && payload.status !== "000") {
      throw new Error(`OpenDART API 오류(${payload.status}): ${payload.message || "알 수 없는 오류"}`);
    }
    return payload;
  } finally {
    clearTimeout(timeout);
  }
}

function readFreshCorpCache(now = Date.now()) {
  try {
    const text = fs.readFileSync(CORP_CACHE_PATH, "utf8");
    const cache = JSON.parse(text);
    const fetchedAt = Date.parse(cache.fetched_at);
    if (
      Number.isFinite(fetchedAt) &&
      now - fetchedAt < CORP_CACHE_TTL_MS &&
      Array.isArray(cache.items)
    ) {
      return cache.items;
    }
  } catch (error) {
    if (error.code !== "ENOENT") {
      console.error(`DART corp code cache read failed: ${error.message}`);
    }
  }
  return null;
}

function writeCorpCache(items) {
  fs.mkdirSync(DATA_DIR, { recursive: true });
  fs.writeFileSync(
    CORP_CACHE_PATH,
    JSON.stringify(
      {
        fetched_at: new Date().toISOString(),
        source: `${DART_BASE_URL}/corpCode.xml`,
        items,
      },
      null,
      2,
    ),
  );
}

function normalizeCorpList(parsed) {
  const rawList = parsed?.result?.list;
  const list = Array.isArray(rawList) ? rawList : rawList ? [rawList] : [];
  return list
    .map((item) => ({
      corp_code: String(item.corp_code ?? "").trim(),
      corp_name: String(item.corp_name ?? "").trim(),
      stock_code: String(item.stock_code ?? "").trim(),
      modify_date: String(item.modify_date ?? "").trim(),
    }))
    .filter((item) => item.corp_code && item.corp_name && item.stock_code);
}

async function downloadCorpCodes() {
  requireDartCredential();
  const url = new URL(`${DART_BASE_URL}/corpCode.xml`);
  url.searchParams.set("crtfc_key", DART_API_KEY);

  const zipBuffer = await fetchWithTimeout(url, { responseType: "arrayBuffer" });
  const zip = new AdmZip(zipBuffer);
  const entry = zip.getEntry("CORPCODE.xml");
  if (!entry) {
    throw new Error("OpenDART corpCode.xml ZIP에 CORPCODE.xml이 없습니다.");
  }

  const xml = entry.getData().toString("utf8");
  const parser = new XMLParser({ ignoreAttributes: false, trimValues: true });
  const items = normalizeCorpList(parser.parse(xml));
  if (items.length === 0) {
    throw new Error("OpenDART 고유번호 파일에서 상장사 목록을 찾지 못했습니다.");
  }
  writeCorpCache(items);
  return items;
}

async function getCorpCodes() {
  return readFreshCorpCache() || await downloadCorpCodes();
}

async function resolveCorp(stockName) {
  const input = normalizeStockInput(stockName);
  if (!input) throw new Error("stock_name 파라미터가 누락되었습니다.");

  const corps = await getCorpCodes();
  const matches = /^\d{6}$/.test(input)
    ? corps.filter((corp) => corp.stock_code === input)
    : corps.filter((corp) => corp.corp_name === input);

  if (matches.length === 0) {
    throw new Error(`'${input}'의 DART 고유번호를 찾을 수 없습니다.`);
  }
  if (matches.length > 1) {
    const candidates = matches.map((corp) => `${corp.corp_name}(${corp.stock_code}, ${corp.corp_code})`).join(", ");
    throw new Error(`'${input}'의 DART 고유번호 매칭이 모호합니다: ${candidates}. 6자리 종목코드를 입력하세요.`);
  }
  return matches[0];
}

function formatDate(date) {
  return date.toISOString().slice(0, 10).replaceAll("-", "");
}

function queryWindow(days = 30) {
  const end = new Date();
  const start = new Date(end);
  start.setDate(start.getDate() - days);
  return { bgn_de: formatDate(start), end_de: formatDate(end) };
}

async function dartJson(pathname, params) {
  requireDartCredential();
  const url = new URL(`${DART_BASE_URL}/${pathname}`);
  url.searchParams.set("crtfc_key", DART_API_KEY);
  for (const [key, value] of Object.entries(params)) {
    if (value !== undefined && value !== null && value !== "") {
      url.searchParams.set(key, String(value));
    }
  }
  return await fetchWithTimeout(url);
}

function asList(value) {
  return Array.isArray(value) ? value : [];
}

async function fetchDisclosureList(corp, window) {
  const rows = [];
  for (const detailType of SHAREHOLDING_DETAIL_TYPES) {
    const payload = await dartJson("list.json", {
      corp_code: corp.corp_code,
      bgn_de: window.bgn_de,
      end_de: window.end_de,
      pblntf_ty: "D",
      pblntf_detail_ty: detailType,
      page_no: 1,
      page_count: 10,
    });
    rows.push(...asList(payload.list).map((row) => ({ ...row, detailType })));
  }
  return rows
    .sort((a, b) => String(b.rcept_dt || "").localeCompare(String(a.rcept_dt || "")))
    .slice(0, 10);
}

async function fetchMajorStock(corp) {
  const payload = await dartJson("majorstock.json", { corp_code: corp.corp_code });
  return asList(payload.list).slice(0, 5);
}

async function fetchExecutiveStock(corp) {
  const payload = await dartJson("elestock.json", { corp_code: corp.corp_code });
  return asList(payload.list).slice(0, 5);
}

function value(row, keys) {
  for (const key of keys) {
    const item = row?.[key];
    if (item !== undefined && item !== null && item !== "") return item;
  }
  return "-";
}

function formatDisclosureRows(rows) {
  if (rows.length === 0) return "- 최근 30일 지분공시 목록이 없습니다.";
  return rows
    .map((row) => {
      const date = value(row, ["rcept_dt"]);
      const report = value(row, ["report_nm"]);
      const receipt = value(row, ["rcept_no"]);
      const submitter = value(row, ["flr_nm"]);
      return `- ${date} | ${report} | 접수번호 ${receipt} | 제출자 ${submitter}`;
    })
    .join("\n");
}

function formatMajorRows(rows) {
  if (rows.length === 0) return "- 5% 룰 대량보유 요약 데이터가 없습니다.";
  return rows
    .map((row) => {
      const reporter = value(row, ["repror", "rpt_nm", "stkqy"]);
      const ratio = value(row, ["stkrt", "posesn_stock_qota_rt"]);
      const change = value(row, ["stkrt_change", "stock_qota_rt_change"]);
      const purpose = value(row, ["hold_stock_purpose", "report_resn"]);
      return `- 보고자: ${reporter} | 보유비율: ${ratio}% | 변동비율: ${change}% | 보유목적: ${purpose}`;
    })
    .join("\n");
}

function formatExecutiveRows(rows) {
  if (rows.length === 0) return "- 임원/주요주주 소유 요약 데이터가 없습니다.";
  return rows
    .map((row) => {
      const reporter = value(row, ["repror", "rpt_nm"]);
      const relation = value(row, ["isu_exctv_rgist_at", "isu_main_shrholdr", "relate"]);
      const quantity = value(row, ["sp_stock_lmp_cnt", "stock_knd"]);
      const change = value(row, ["sp_stock_lmp_irds_cnt", "irds_cnt"]);
      return `- 보고자: ${reporter} | 관계/구분: ${relation} | 소유주식수: ${quantity} | 증감: ${change}`;
    })
    .join("\n");
}

async function getDisclosureSignal(stockName) {
  const corp = await resolveCorp(stockName);
  const window = queryWindow(30);
  const [disclosures, majorRows, executiveRows] = await Promise.all([
    fetchDisclosureList(corp, window),
    fetchMajorStock(corp),
    fetchExecutiveStock(corp),
  ]);

  return `
[${corp.corp_name}] DART 지분공시 signal
- 종목코드: ${corp.stock_code}
- 고유번호: ${corp.corp_code}
- 조회기간: ${window.bgn_de}~${window.end_de}

[최신 공시]
${formatDisclosureRows(disclosures)}

[5% 룰 대량보유 요약]
${formatMajorRows(majorRows)}

[임원 주요주주 거래 요약]
${formatExecutiveRows(executiveRows)}
  `.trim();
}

      inputSchema: {
        type: "object",
        properties: {
          stock_name: {
            type: "string",

server.setRequestHandler(CallToolRequestSchema, async (request) => {

const transport = new StdioServerTransport();
await server.connect(transport);

console.error("DART MCP Server is running...");
```
